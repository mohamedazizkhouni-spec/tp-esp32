03 Notions théoriques

⚡ La résistance de la LED

Sans résistance, le courant devient trop fort et détruit la LED. Loi d’Ohm : R = (V_alim − V_LED) / I. Essaie avec tes valeurs :

R minimale = 130 Ω (on prend 220 Ω environ)

L’écran OLED communique avec seulement deux fils.

04 Expérience 1 : allumer une LED

La LED est branchée sur la GPIO 2 (LED_BUILTIN) à travers une résistance, l’autre patte au GND. Toutes les secondes, elle change d’état et l’écran suit.

FilDeVers
🟣 VioletGPIO 2LED (via résistance)
⚫ NoirGNDAutre patte de la LED

Le code

#include <WiFi.h>
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define LED_BUILTIN 2

Adafruit_SSD1306 oled(128, 64, &Wire, -1);

void setup() {
  Serial.begin(115200);
  pinMode(LED_BUILTIN, OUTPUT);
  Wire.begin(21, 22);
  oled.begin(SSD1306_SWITCHCAPVCC, 0x3C);
  oled.setTextColor(SSD1306_WHITE);
  WiFi.mode(WIFI_STA);
  delay(500);
}

void loop() {
  // --- LED ALLUMEE ---
  digitalWrite(LED_BUILTIN, HIGH);
  oled.clearDisplay();
  oled.setCursor(0, 0);
  oled.setTextSize(1);
  oled.println("--- Etat LED ---");
  oled.setTextSize(2);
  oled.println("LED ALLUMEE");
  oled.display();
  delay(1000);

  // --- LED ETEINTE ---
  digitalWrite(LED_BUILTIN, LOW);
  oled.clearDisplay();
  oled.setCursor(0, 0);
  oled.setTextSize(1);
  oled.println("--- Etat LED ---");
  oled.setTextSize(2);
  oled.println("LED ETEINTE");
  oled.display();
  delay(1000);
}

05 Expérience 2 : feux de carrefour

Trois LEDs, chacune avec sa résistance. Une seule est allumée à la fois et l’écran affiche la couleur et le temps restant. Lance la simulation :

🟡 JauneGPIO 262 s

Chaque LED a sa broche et sa résistance, avec le GND commun.

3

Le code

/*
  TP ESP32 - Feux de carrefour avec écran OLED
  Carte : TTGO ESP32 (OLED SSD1306 128x64, I2C : SDA = 21, SCL = 22)

  Câblage (chaque LED en série avec une résistance de ~220 ohms) :
    LED rouge -> GPIO 25
    LED jaune -> GPIO 26
    LED verte -> GPIO 27
    L'autre patte de chaque résistance/LED -> GND
*/

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

// ----- Broches des LEDs -----
#define LED_ROUGE 25
#define LED_JAUNE 26
#define LED_VERT  27

// ----- Durées des phases (en secondes) -----
#define DUREE_ROUGE 5
#define DUREE_VERT  5
#define DUREE_JAUNE 2

Adafruit_SSD1306 oled(128, 64, &Wire, -1);

// Allume une seule LED à la fois
void allumer(bool rouge, bool jaune, bool vert) {
  digitalWrite(LED_ROUGE, rouge);
  digitalWrite(LED_JAUNE, jaune);
  digitalWrite(LED_VERT,  vert);
}

// Affiche la couleur active et le temps restant sur l'écran OLED
void afficher(const char* couleur, int restant) {
  oled.clearDisplay();

  oled.setTextSize(1);
  oled.setCursor(0, 0);
  oled.println("--- FEU CARREFOUR ---");

  oled.setTextSize(3);              // Couleur en grand
  oled.setCursor(0, 22);
  oled.println(couleur);

  oled.setTextSize(1);
  oled.setCursor(0, 54);
  oled.print("Reste : ");
  oled.print(restant);
  oled.println(" s");

  oled.display();
}

// Exécute une phase : allume la LED, puis compte à rebours à l'écran
void phase(bool rouge, bool jaune, bool vert, const char* couleur, int duree) {
  allumer(rouge, jaune, vert);
  for (int t = duree; t > 0; t--) {
    afficher(couleur, t);
    delay(1000);
  }
}

void setup() {
  Serial.begin(115200);

  pinMode(LED_ROUGE, OUTPUT);
  pinMode(LED_JAUNE, OUTPUT);
  pinMode(LED_VERT,  OUTPUT);
  allumer(false, false, false);

  Wire.begin(21, 22);
  if (!oled.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("Erreur : ecran OLED introuvable");
    while (true);                   // On bloque si l'écran n'est pas détecté
  }
  oled.setTextColor(SSD1306_WHITE);
}

void loop() {
  phase(true,  false, false, "ROUGE", DUREE_ROUGE);
  phase(false, false, true,  "VERT",  DUREE_VERT);
  phase(false, true,  false, "JAUNE", DUREE_JAUNE);
}

06 Difficultés rencontrées

🔦LED toujours allumée

Ajout de digitalWrite(LED_BUILTIN, LOW) dans la partie « éteinte ».

🖥️Écran noir

Vérification de l’adresse 0x3C, des broches SDA/SCL et de oled.display().

🔥Risque pour les LEDs

Une résistance en série avec chaque LED.
